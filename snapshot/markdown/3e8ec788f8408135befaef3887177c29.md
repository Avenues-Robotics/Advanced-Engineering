    public void drive(double speed, int ticks){

        if (opModeIsActive()) {
        
            // Add comment

            motorRight.setMode(DcMotor.RunMode.STOP_AND_RESET_ENCODER);

            motorRight.setTargetPosition(ticks);
            
            // Add comment
            motorLeft.setMode(DcMotor.RunMode.RUN_TO_POSITION);
